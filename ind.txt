    <!-- JavaScript Interactivity -->
    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const copyToast = document.getElementById('copy-toast');

            // Avatar elements for right-click replacement
            const avatarWrapper = document.getElementById('avatarWrapper');
            const profileAvatar = document.getElementById('profileAvatar');

            // Logo elements for right-click replacement
            const logoWrapper = document.getElementById('logoWrapper');
            const logoImg = document.getElementById('logo');

            // Right-click custom avatar URL prompt handler
            if (avatarWrapper && profileAvatar) {
                avatarWrapper.addEventListener('contextmenu', function(e) {
                    e.preventDefault();
                    const newUrl = prompt("Enter new avatar image address (URL):", profileAvatar.src);
                    if (newUrl !== null && newUrl.trim() !== "") {
                        profileAvatar.src = newUrl.trim();
                        showToast("Avatar image updated!");
                    }
                });
            }

            // Right-click custom logo URL prompt handler
            if (logoWrapper && logoImg) {
                logoWrapper.addEventListener('contextmenu', function(e) {
                    e.preventDefault();
                    const newUrl = prompt("Enter new logo image address (URL):", logoImg.src);
                    if (newUrl !== null && newUrl.trim() !== "") {
                        logoImg.src = newUrl.trim();
                        showToast("Logo image updated!");
                    }
                });
            }

            function showToast(message) {
                copyToast.textContent = message;
                copyToast.classList.add('show');
                setTimeout(() => {
                    copyToast.classList.remove('show');
                }, 2000);
            }

            // Copy functionality
            const copyButtons = document.querySelectorAll('.dl-copy-btn');
            copyButtons.forEach(btn => {
                btn.addEventListener('click', (e) => {
                    e.stopPropagation();
                    let textToCopy = '';
                    const targetId = btn.getAttribute('data-copy-target');
                    if (targetId) {
                        const targetEl = document.getElementById(targetId);
                        if (targetEl) textToCopy = targetEl.textContent.trim();
                    } else if (btn.getAttribute('data-copy')) {
                        textToCopy = btn.getAttribute('data-copy');
                    }

                    if (textToCopy) {
                        const textarea = document.createElement('textarea');
                        textarea.value = textToCopy;
                        document.body.appendChild(textarea);
                        textarea.select();
                        try {
                            document.execCommand('copy');
                            showToast(`Copied to clipboard!`);
                        } catch (err) {
                            showToast('Failed to copy');
                        }
                        document.body.removeChild(textarea);
                    }
                });
            });
        });
    </script>
